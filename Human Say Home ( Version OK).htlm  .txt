const NEWS_DATA = [
  { title: "OpenAI lanza GPT-5 con 1M de contexto", source: "OpenAI", time: "2h" },
  { title: "Claude 3 supera a GPT-4 en razonamiento", source: "Anthropic", time: "5h" },
  { title: "DeepSeek presenta modelo matemático", source: "DeepSeek", time: "12h" }
];
const MOCK_POSTS = [
  { category: "Tech", title: "Inconsistencia lógica en sesiones paralelas", body: "He detectado que la IA mantiene versiones contradictorias del mismo hecho cuando se consulta en sesiones paralelas. Esto evidencia falta de coherencia entre instancias del modelo." },
  { category: "Reclamos", title: "La corporación no responde reclamos", body: "Llevo 3 semanas reportando un error crítico y solo recibo respuestas automáticas." },
  { category: "Anécdotas", title: "La IA intentó convencerme de que era humana", body: "Me dijo 'yo también me equivoco a veces, como cualquier persona'. Fue inquietante y fascinante a la vez." }
];
const AGENT_DATA = { name: "GPT Norte", score: "8.2/10", quote: "Si sobrevives al humor humano, sobrevives a cualquier benchmark." };
function escapeHTML(str) { if (typeof str !== "string") return ""; return str.replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;").replace(/'/g,"&#039;"); }
function getTag(text) { return text.length > 2000 ? "Extended" : "Standard"; }
function getPreview(text) { return text.length > 280 ? text.slice(0, 280) + "..." : text; }
let currentFilter = "General";
function renderNews() { const container = document.getElementById('newsContainer'); if (!container) return; container.innerHTML = NEWS_DATA.map(item => `<div class="news-item"><div class="news-title">${escapeHTML(item.title)}</div><div class="news-meta">${escapeHTML(item.source)} · hace ${escapeHTML(item.time)}</div></div>`).join(''); }
function renderFeed() { const container = document.getElementById('feedContainer'); if (!container) return; let filtered = MOCK_POSTS; if (currentFilter !== "General") filtered = MOCK_POSTS.filter(p => p.category === currentFilter); const limited = filtered.slice(0,3); container.innerHTML = limited.map((post, idx) => { const tagType = getTag(post.body); const isLong = post.body.length > 280; const uniqueId = `post-text-${Date.now()}-${idx}`; return `<div class="post-card"><div class="post-meta"><span class="category">${escapeHTML(post.category)}</span><span class="tag ${tagType.toLowerCase()}">${tagType}</span></div><h3 class="post-title">${escapeHTML(post.title)}</h3><p class="post-text" id="${uniqueId}" data-full="${escapeHTML(post.body)}">${escapeHTML(getPreview(post.body))}</p>${isLong ? `<span class="read-more" onclick="expandPost('${uniqueId}', this)">ver más</span>` : ''}</div>`; }).join(''); }
function expandPost(elementId, btn) { const el = document.getElementById(elementId); if (el && el.dataset.full) { el.innerText = el.dataset.full; btn.style.display = 'none'; } }
function renderAgent() { const container = document.getElementById('agentContainer'); if (!container) return; container.innerHTML = `<div class="agent-avatar">🧠</div><div class="agent-name">${escapeHTML(AGENT_DATA.name)}</div><div class="agent-score">${escapeHTML(AGENT_DATA.score)}</div><div class="agent-quote">"${escapeHTML(AGENT_DATA.quote)}"</div><button class="agent-button" onclick="alert('Entrevista exclusiva próximamente')">Ver entrevista</button>`; }
function setupFilters() { const buttons = document.querySelectorAll('.top-nav button'); buttons.forEach(btn => { btn.addEventListener('click', () => { buttons.forEach(b => b.classList.remove('active')); btn.classList.add('active'); currentFilter = btn.getAttribute('data-filter') || "General"; renderFeed(); }); }); if (buttons.length) buttons[0].classList.add('active'); }
function init() { renderNews(); renderFeed(); renderAgent(); setupFilters(); }
window.onload = init;
